```bash
# On hub box (16.54.141.121, ubuntu user, has VPC + PG16 + 12-CPU + fast disk)

#!/bin/bash
PGPASSWORD="GuZbKkezYSzX4qGTxnkTynH0AIJy" pg_dump \
  -h novakash-pg-prod.cpmisy2asv71.ca-central-1.rds.amazonaws.com \
  -U postgres -d novakash \
  -t strategy_decisions -t market_data -t trades -t signal_evaluations \
  -t window_snapshots -t window_evaluation_traces \
  --data-only --format=c \
  --where="created_at > NOW() - INTERVAL '"'"'30 days'"'"'" \
  > /tmp/novakash_30d_$(date +%Y%m%d).dump 2> /tmp/dump.log
EOF
chmod +x /tmp/run_dump.sh && nohup /tmp/run_dump.sh & echo $!'
# scp the dump back to local
scp -i /tmp/novakash-infra-v2.pem ubuntu@16.54.141.121:/tmp/novakash_30d_*.dump ~/Downloads/
# Local restore (Postgres@15 is fine)
pg_restore -d novakash_local --no-owner --no-privileges -j 2 ~/Downloads/novakash_30d_*.dump
```

-- All current LIVE overrides
SELECT strategy_id, jsonb_pretty(params)
FROM strategy_runtime_overrides
WHERE strategy_id IN ('v12_lgb_combo','v9_1_lgb_only','v_v12_extreme_dn_btc_5m')
ORDER BY strategy_id;

-- Last 24h trade summary
SELECT
  strategy_id, direction,
  COUNT(*) AS n,
  SUM(CASE WHEN outcome='WIN' THEN 1 ELSE 0 END) AS wins,
  ROUND(100.0 * SUM(CASE WHEN outcome='WIN' THEN 1 ELSE 0 END)::numeric / COUNT(*), 1) AS wr,
  ROUND(SUM(pnl_usd)::numeric, 2) AS pnl_db
FROM trades
WHERE created_at > NOW() - INTERVAL '24 hours'
  AND strategy_id IS NOT NULL
GROUP BY 1, 2 ORDER BY n DESC;
